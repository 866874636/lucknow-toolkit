import socket

# 1. User se IP lena
target = input("Target IP daalo (ex: 127.0.0.1): ")

print(f"\n[ scanning {target} ]\n")

# 2. Port 1 se 100 tak check karna
for port in range(1, 102):
    # Socket banao
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s.settimeout(0.5) # 0.5 sec me jawab nahi to band

    # Connect karke dekho
    result = s.connect_ex((target, port))
    
    if result == 0:
        print(f" -> Port {port} : KHULA HAI [OPEN]")
    
    s.close()

print("\n[ Scan Khatam ]")
